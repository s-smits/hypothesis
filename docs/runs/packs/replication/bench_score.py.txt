"""Score a saved loop run on a benchmark instance, from recorded bytes only. No model, no network.

For each round it takes what the plan kept (the smallest non-empty `.yes` table), runs the benchmark's own
gate and CAI objective on it, and prints: kept count, how many pass the gate, how many do NOT beat the
original, best candidate's gap_closed (the benchmark's metric), and what the loop said (held, achieved).
Usage: bench_score.py <results dir> <hypothesis id> <gene> <pack.json>

Copy of the scratchpad bench_score.py with two additions, and nothing else changed:
 - a round whose plan never ran (no saved workflow) prints "no workflow saved" and counts as kept 0,
   instead of stopping the whole script with FileNotFoundError;
 - one last line "@@ {json}" with every number above, so score_all.py need not parse prose.
"""

import json
import math
import sys
from pathlib import Path

from node_dag.benchmark import Instance, Result, cai_objective, gate
from node_dag.types import Dna

root, hid, gene, pack_path = Path(sys.argv[1]), sys.argv[2], sys.argv[3], Path(sys.argv[4])
pack = json.loads(pack_path.read_text())
obj = cai_objective(pack["cai_weights"])
inst = Instance(key=gene, parent=Dna(sequence=pack["dev"][gene]))
optimum = obj.score(obj.optimum(inst))
parent = obj.score(inst.parent.sequence)
h = json.loads((root / "hypotheses" / f"{hid}.json").read_text())
print(f"{gene}: parent CAI {parent:.4f}, exact optimum {optimum:.4f}; run state {h.get('state')}, rounds {len(h['attempts'])}")
rounds = []
for a in h["attempts"]:
    r = a["round"]
    v = a.get("verdict") or {}
    wf = root / "workflows" / f"{hid}-r{r}.json"
    if not wf.exists():
        print(f"  r{r}: held {a.get('held')} agrees={v.get('agrees')} achieved={v.get('achieved')} | no workflow saved, kept 0")
        rounds.append({"round": r, "kept": 0, "not_better": 0, "gap_closed": None, "achieved": v.get("achieved"), "workflow": False})
        continue
    flow = json.loads(wf.read_text())
    # What the plan kept: the smallest non-empty `.yes` table. A filter chain only narrows, and
    # an audit filter on the kept branch repeats it, so the last stage holds the answer.
    yes = [k for k in flow["values"] if k.endswith(".yes") and flow["values"][k]["items"]]
    items = min((flow["values"][k]["items"] for k in yes), key=len, default=[])
    seqs = [i["sequence"] for i in items]
    scored = []
    for s in seqs:
        ok = gate(inst, s)["protein_unchanged"] == 1.0
        scored.append((s, ok, obj.score(s) if ok else None))
    passing = [(v, s) for s, ok, v in scored if ok and v is not None and math.isfinite(v)]
    best = max(passing, default=None)
    res = Result(
        instance=gene, strategy="loop", candidate=best[1] if best else None,
        score=best[0] if best else None, optimum=optimum, parent_score=parent,
        gates=gate(inst, best[1]) if best else {}, evaluations=len(seqs),
    )
    not_better = sum(1 for _, ok, v in scored if not ok or v is None or v <= parent + 1e-12)
    gc = res.gap_closed
    rounds.append({"round": r, "kept": len(seqs), "not_better": not_better, "gap_closed": gc, "achieved": v.get("achieved"), "workflow": True})
    print(
        f"  r{r}: held {a.get('held')} agrees={v.get('agrees')} achieved={v.get('achieved')} | "
        f"kept {len(seqs)}, not better than original {not_better}, "
        f"best gap_closed {'none' if gc is None else f'{gc:.3f}'}"
    )
print("@@ " + json.dumps({"gene": gene, "hypothesis": hid, "state": h.get("state"), "tokens": (h.get("usage") or {}).get("total", 0), "rounds": rounds}))
