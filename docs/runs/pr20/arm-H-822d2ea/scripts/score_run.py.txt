#!/usr/bin/env python3
"""score_run.py <results dir> [--json]: a code-only scorer for a loop results directory.

No model, no network, nothing written into the results directory. For every saved
Hypothesis (hypotheses/*.json, workflows/<id>-rN.json, ledger.jsonl, trajectories) it
prints, per run and per accepted round:

  * state, rounds, tokens (run total from the Hypothesis `usage`, per round from the saved
    trajectories where they carry usage);
  * SCORE CHECK: the kept entities that are strictly better than the FIRST input in each
    non-gate filter column of the kept lineage (direction from the filter: at_least and
    beats_reference(higher) and top_k(largest) count "above", at_most and
    beats_reference(higher=false) count "below"). Ties count as not better. Prints the
    count and the lowest kept value (highest for a "below" filter). PASS needs every
    measurable column clean, FAIL says how many kept do not beat the first input,
    UNMEASURABLE says why (no outcome saved, no non-gate filter, first input never
    scored in that column, kept entity unscored), EMPTY is an accept that kept nothing;
  * STOP CHECK: for DNA kept entities, how many have a different stop codon than the
    parent they came from. The parent is recovered by re-running the recorded
    mutate_synonymous / resample_synonymous / domesticate / gc_target_recode /
    codon_optimise / recode_targeted config on the saved input table, in a worker process
    over each candidate code export, and accepted only when the re-run reproduces the
    recorded output table (so the code version is chosen by bytes, not assumed).
    'unmeasurable' where it does not reproduce or no recoder produced the entity;
  * GATE: for benchmark gene goals (the CAI goal over a dev gene: tusE, aroE, arfA; any
    other gene is NOT scored, so the held-out split is never opened) the immutable-honouring
    gate of PR #14 (node_dag.benchmark.gate / passes) on every kept sequence, the parent
    being the first input.

and then the denominators: launched, accepted by verdict, pass the score check, pass the
score check AND keep every parent's stop codon (and, separately, pass the gate).

Usage:
  score_run.py <dir> [--json | --brief] [--rounds accepted|all] [--no-rerun] [--src EXPORT ...]
  <dir> is a results directory (hypotheses/, workflows/, ledger.jsonl), or any directory
  above several (the archive docs/runs): every descendant with a hypotheses/ is read.
  --src adds a code export to try (a git archive of a commit); the defaults are the
  exports under $SCORE_RUN_EXPORTS (default <this dir>/triage-5/<sha7>) of 822d2ea, 56acba2,
  84dd787, be3805f (compact head, has recode_codons), extracted on demand from $SCORE_RUN_REPO with git archive.
  Run with the loop's venv python, e.g.
    /Users/air/Developer/hypothesis-loop-merged/.venv/bin/python score_run.py <dir>
Files are read defensively (a live worker may be writing): bad JSON or a missing key
degrades that row to 'unreadable' / 'unmeasurable', never an exception.
"""

def jload(path):
    """(object, None) or (None, reason). Never raises."""
    try:
        return json.loads(Path(path).read_text()), None
    except FileNotFoundError:
        return None, "missing"
    except (OSError, ValueError) as e:
        return None, f"unreadable ({type(e).__name__})"


def g(d, *keys, default=None):
    """d[k1][k2]... where any step may be missing or the wrong type."""
    for k in keys:
        if isinstance(d, dict):
            d = d.get(k)
        elif isinstance(d, list) and isinstance(k, int) and -len(d) <= k < len(d):
            d = d[k]
        else:
            return default
        if d is None:
            return default
    return d


def find_result_dirs(root: Path):
    root = Path(root)
    if (root / "hypotheses").is_dir():
        return [root]
    out = []
    for dp, dns, _ in os.walk(root):
        depth = len(Path(dp).relative_to(root).parts)
        if "hypotheses" in dns:
            out.append(Path(dp))
            dns[:] = []
        elif depth >= 5:
            dns[:] = []
    return sorted(out)


def load_index(root: Path):
    """hypothesis id -> {kind, commit} from the archive's index.json, when there is one."""
    for d in [Path(root), *Path(root).resolve().parents][:4]:
        data, _ = jload(d / "index.json")
        if isinstance(data, dict) and isinstance(data.get("runs"), list):
            return {
                r.get("hypothesis_id"): r for r in data["runs"] if isinstance(r, dict)
            }
    return {}


def read_ledger(d: Path):
    out = {}
    p = d / "ledger.jsonl"
    if not p.exists():
        return out
    try:
        lines = p.read_text().splitlines()
    except OSError:
        return out
    for line in lines:
        try:
            row = json.loads(line)
        except ValueError:
            continue  # a half-written last line
        if isinstance(row, dict) and row.get("hypothesis"):
            out[row["hypothesis"]] = row
    return out


def msg_tokens(msgs):
    t = 0
    for m in msgs if isinstance(msgs, list) else []:
        u = m.get("usage") if isinstance(m, dict) else None
        if isinstance(u, dict):
            t += int(u.get("input_tokens") or 0) + int(u.get("output_tokens") or 0)
    return t


def round_tokens(d: Path, hid: str, rnd) -> int | None:
    """input+output tokens of one round's plan, verify and critique trajectories."""
    total, seen = 0, False
    for kind in ("plan", "verify", "critique"):
        p = d / "trajectories" / f"{hid}-r{rnd}-{kind}.json"
        if p.exists():
            data, _ = jload(p)
            if data is not None:
                total += msg_tokens(data)
                seen = True
    gz = d / "trajectories" / f"{hid}.jsonl.gz"
    if not seen and gz.exists():
        try:
            with gzip.open(gz, "rt") as fh:
                for line in fh:
                    try:
                        row = json.loads(line)
                    except ValueError:
                        continue
                    if re.search(rf"-r{rnd}-(plan|verify|critique)\.json$", row.get("file", "")):
                        total += msg_tokens(row.get("data"))
                        seen = True
        except (OSError, EOFError):
            pass
    return total if seen else None


# --------------------------------------------------------------------- outcome helpers


def attempt_outcome(d: Path, att: dict, hid=None):
    """(dag, values, note): the saved workflow file when there is one, else the embedded outcome."""
    wid = att.get("workflow_id")
    if not wid and hid and att.get("round") is not None and (d / "workflows" / f"{hid}-r{att.get('round')}.json").exists():
        wid = f"{hid}-r{att.get('round')}"  # the attempt did not record its workflow id, the file is named by the usual rule
    if wid:
        w, why = jload(d / "workflows" / f"{wid}.json")
        if isinstance(w, dict) and isinstance(w.get("values"), dict):
            return (w.get("dag") or att.get("dag") or {}), w["values"], None
        note = f"workflow file {why}"
    else:
        note = "no workflow id (no DAG run saved for this round)"
    vals = g(att, "outcome", "values")
    if isinstance(vals, dict) and vals:
        return (att.get("dag") or {}), vals, None
    return (att.get("dag") or {}), {}, note


def steps_of(dag):
    s = g(dag, "steps", default={})
    return s if isinstance(s, dict) else {}


def base_of(source: str) -> str:
    return source.split(".")[0]


def consumed(dag):
    used = set()
    for st in steps_of(dag).values():
        ins = g(st, "inputs", default={})
        if isinstance(ins, dict):
            used.update(v for v in ins.values() if isinstance(v, str))
        ref = g(st, "config", "scored_in")
        if isinstance(ref, str):
            used.add(ref)
    return used


def items_of(values, key):
    t = values.get(key)
    it = t.get("items") if isinstance(t, dict) else None
    if not isinstance(it, list):
        return []
    return [x for x in it if isinstance(x, dict) and isinstance(x.get("sequence"), str)]


def kept_tables(dag, values, plan):
    """Tables that hold what the plan kept. Returns (keys, how)."""
    res = g(plan, "result_source")
    if isinstance(res, str) and items_of(values, res):
        return [res], "plan.result_source"
    used = consumed(dag)
    ys = [k for k in values if k.endswith(".yes")]
    if ys:
        return [k for k in ys if k not in used and items_of(values, k)], "unconsumed .yes"
    st = steps_of(dag)
    return [k for k in st if k not in used and k in values and items_of(values, k)], "unconsumed step"


def kept_entities(dag, values, plan):
    keys, how = kept_tables(dag, values, plan)
    seen, out = set(), []
    for k in keys:
        for it in items_of(values, k):
            eid = it.get("id") or it["sequence"]
            if eid not in seen:
                seen.add(eid)
                out.append(it)
    return out, keys, how


def score_index(values):
    idx = {}
    for t in values.values():
        sc = t.get("scores") if isinstance(t, dict) else None
        if not isinstance(sc, dict):
            continue
        for col, per in sc.items():
            if isinstance(per, dict):
                m = idx.setdefault(col, {})
                for eid, v in per.items():
                    m.setdefault(eid, v)
    return idx


def first_input(h, values, dag):
    inp = h.get("inputs")
    if isinstance(inp, dict) and inp:
        v = inp[next(iter(inp))]
        v = v if isinstance(v, list) else [v]
        if v and isinstance(v[0], dict) and "sequence" in v[0]:
            e = dict(v[0])
            e.setdefault("id", hashlib.sha256(f"{e.get('kind', 'dna')}:{e['sequence']}".encode()).hexdigest()[:12])
            return e
    keys = list(g(dag, "inputs", default={}) or {})
    if keys and items_of(values, keys[0]):
        return items_of(values, keys[0])[0]
    return None


def short_col(c: str) -> str:
    return c.split("__")[-1] if "__" in c else c


def lineage(dag, start_keys):
    """step keys that feed the kept tables, including those tables' own steps."""
    st = steps_of(dag)
    todo, seen = [base_of(k) for k in start_keys], set()
    while todo:
        k = todo.pop()
        if k in seen:
            continue
        seen.add(k)
        ins = g(st.get(k), "inputs", default={})
        if isinstance(ins, dict):
            todo += [base_of(v) for v in ins.values() if isinstance(v, str)]
    return seen


def fnum(x):
    try:
        v = float(x)
    except (TypeError, ValueError):
        return None
    return v if math.isfinite(v) else None


_FIRST = re.compile(r"\bfirst (?:input )?(?:sequence|input|entity)\b|\bthe first\b")
_UP = re.compile(r"\b(?:higher|greater|more|above|increas\w*|rais\w*|exceed\w*|outscor\w*)\b")
_DOWN = re.compile(r"\b(?:fewer|lower|less|smaller|below|decreas\w*|reduc\w*)\b")


def toward_first(claim) -> int:
    """+1 when the claim says higher than the first input, -1 lower, 0 when it is no first-input comparison.

    The same rule as accept-code's replay (work/proposal.py), so the two agree on every recorded claim."""
    if not isinstance(claim, str):
        return 0
    t = claim.lower()
    m = _FIRST.search(t)
    if not m:
        return 0
    before = [(w.start(), sgn) for pat, sgn in ((_UP, 1), (_DOWN, -1)) for w in pat.finditer(t[: m.start()])]
    if before:
        return max(before)[1]
    after = [(w.start(), sgn) for pat, sgn in ((_UP, 1), (_DOWN, -1)) for w in pat.finditer(t[m.end():])]
    return min(after)[1] if after else 0


def filter_up(cfg):
    name = cfg.get("name")
    if name == "at_least":
        return True
    if name == "at_most":
        return False
    if name == "beats_reference":
        return bool(cfg.get("higher", True))
    if name == "top_k":
        return bool(cfg.get("largest", True))
    return None


def col_pairs(values_list, col):
    """{sequence: score} for one score column over the saved tables."""
    out = {}
    for values in values_list:
        seq = {}
        for t in values.values():
            for it in (t.get("items") if isinstance(t, dict) else None) or []:
                if isinstance(it, dict) and isinstance(it.get("sequence"), str):
                    seq[it.get("id")] = it["sequence"]
        for t in values.values():
            per = g(t, "scores", col)
            if isinstance(per, dict):
                for eid, v in per.items():
                    if eid in seq and fnum(v) is not None:
                        out[seq[eid]] = fnum(v)
    return out


def derive_baseline(pairs, seq0):
    """Score of seq0 when the column is an exact linear function of the A/C/G/T counts over the saved pairs.

    Only when the fit is full rank (4 weights) from at least 6 distinct sequences and every residual is below
    1e-6 (an atom count is such a function; OSTIR and CAI are not, and fail the residual test). Returns
    (value, n_pairs, max_residual) or None.
    """
    rows = [([sq.count(b) for b in "ACGT"], v) for sq, v in pairs.items() if not set(sq) - set("ACGT")]
    if len(rows) < 6 or not isinstance(seq0, str) or set(seq0) - set("ACGT"):
        return None
    n = 4
    m = [[sum(r[i] * r[j] for r, _ in rows) for j in range(n)] + [sum(r[i] * v for r, v in rows)] for i in range(n)]
    for c in range(n):
        piv = max(range(c, n), key=lambda i: abs(m[i][c]))
        if abs(m[piv][c]) < 1e-9:
            return None
        m[c], m[piv] = m[piv], m[c]
        for i in range(n):
            if i != c:
                f = m[i][c] / m[c][c]
                m[i] = [a - f * b for a, b in zip(m[i], m[c])]
    w = [m[i][n] / m[i][i] for i in range(n)]
    resid = max(abs(sum(wi * ri for wi, ri in zip(w, r)) - v) for r, v in rows)
    if resid > 1e-6:
        return None
    return sum(wi * seq0.count(b) for wi, b in zip(w, "ACGT")), len(rows), resid


def score_check(h, plan, dag, values, kept, kept_keys, other_values=None):
    """Kept entities strictly better than the first input, per score column of a non-gate filter.

    A column is COUNTED toward the verdict when a criterion whose claim compares with the first input
    ("higher than that of the first input", "fewer than the first") has an assertion on the filter that
    owns it; the direction is the claim's. If no such assertion sits on a filter, the non-gate filters in
    the kept lineage that point the claim's way are counted instead (via='lineage'). Every other non-gate
    filter column in the lineage is listed but not counted: it is a bar, not a comparison with the first
    input (for example amino_acid_changes at most 0, which the first input itself meets by being 0).
    """
    fi = first_input(h, values, dag)
    idx = score_index(values)
    st = steps_of(dag)
    crit = [c for c in (h.get("criteria") or []) if isinstance(c, dict)]
    claim_dir = {c.get("id"): toward_first(c.get("claim")) for c in crit}
    asserts = [a for a in (g(plan, "assertions", default=[]) or []) if isinstance(a, dict)]
    lin = lineage(dag, kept_keys)

    def col_of(step):
        cfg = g(st.get(step), "config", default={})
        col = cfg.get("column")
        if cfg.get("name") in FILTERS and isinstance(col, str) and col and not col.startswith("constraint_check"):
            return cfg, col
        return None, None

    counted = {}  # step -> (direction up?, via)
    for a in asserts:
        d = claim_dir.get(a.get("criterion"), 0)
        cfg, col = col_of(a.get("step"))
        if d and col and a["step"] not in counted:
            counted[a["step"]] = (d > 0, "assertion")
    if not counted:
        for k in sorted(lin):
            cfg, col = col_of(k)
            if col is None:
                continue
            for d in {x for x in claim_dir.values() if x}:
                if (d > 0) == filter_up(cfg) and k not in counted:
                    counted[k] = (d > 0, "lineage")
    cols, seen = [], set()
    # counted steps first, so an info row for the same column and direction is dropped
    order = sorted(counted) + sorted(k for k in lin if col_of(k)[1] and k not in counted)
    for k in order:
        cfg, col = col_of(k)
        if col is None:
            continue
        is_counted = k in counted
        up, via = counted.get(k, (filter_up(cfg), "info"))
        if (col, up) in seen:
            continue
        seen.add((col, up))
        base = fnum(g(idx, col, fi["id"])) if fi else None
        base_from = "this round" if base is not None else None
        if base is None and fi and other_values is not None:
            # entity ids are content hashes and the column name carries the scorer's config hash, so another
            # round of the same run that scored the first input in this column holds the same number
            for ov in other_values():
                base = fnum(g(score_index(ov), col, fi["id"]))
                if base is not None:
                    base_from = "another round of this run"
                    break
        if base is None and fi:
            pairs = col_pairs([values] + (list(other_values()) if other_values else []), col)
            got = derive_baseline(pairs, fi.get("sequence"))
            if got:
                base, base_from = round(got[0], 9), f"derived: exact linear fit on A/C/G/T counts over {got[1]} saved pairs, max residual {got[2]:.1e}"
        have = [v for v in (fnum(g(idx, col, e.get("id"))) for e in kept) if v is not None]
        row = dict(step=k, filter=cfg.get("name"), column=short_col(col), up=up, counted=is_counted, via=via,
                   baseline=base, baseline_from=base_from, n_kept=len(kept), n_scored=len(have), n_not_better=None, edge=None)
        if base is not None:
            row["n_not_better"] = sum(1 for v in have if not (v > base if up else v < base))
            row["edge"] = (min(have) if up else max(have)) if have else None
        cols.append(row)
    return cols, any(claim_dir.values())


def verdict_of_score(cols, any_first_claim, n_kept, have_outcome, note):
    if not have_outcome:
        return "UNMEASURABLE", note or "no outcome saved"
    if n_kept == 0:
        return "EMPTY", "accepted with nothing kept"
    cnt = [c for c in cols if c["counted"]]
    meas = [c for c in cnt if c["baseline"] is not None]
    bad = [c for c in meas if c["n_not_better"]]
    if bad:
        c = bad[0]
        return "FAIL", f"{c['n_not_better']} of {c['n_kept']} kept not strictly {'above' if c['up'] else 'below'} the first input's {c['baseline']:.10g} in {c['column']}"
    if not any_first_claim:
        return "UNMEASURABLE", "no criterion compares with the first input"
    if not cnt:
        return "UNMEASURABLE", "no non-gate filter carries a first-input criterion"
    if not meas:
        return "UNMEASURABLE", f"first input not scored in {cnt[0]['column']} (no saved table holds it)"
    short = [c for c in meas if c["n_scored"] < c["n_kept"]]
    if short:
        return "UNMEASURABLE", f"{short[0]['n_kept'] - short[0]['n_scored']} kept entities unscored in {short[0]['column']}"
    extra = f" ({len(meas)} of {len(cnt)} counted columns measurable)" if len(meas) < len(cnt) else ""
    if any(str(c.get("baseline_from", "")).startswith("derived") for c in meas):
        extra += " [first input's score derived, not saved]"
    return "PASS", "all kept strictly better" + extra


# --------------------------------------------------------------------------- the worker


def ensure_export(sha: str):
    p = EXPORTS / sha
    if (p / "src" / "node_dag").is_dir():
        return p
    try:
        p.mkdir(parents=True, exist_ok=True)
        git = subprocess.Popen(["git", "-C", REPO, "archive", sha], stdout=subprocess.PIPE)
        subprocess.run(["tar", "-x", "-C", str(p)], stdin=git.stdout, check=True)
        git.wait()
    except (OSError, subprocess.SubprocessError):
        return None
    return p if (p / "src" / "node_dag").is_dir() else None


def run_worker(export: Path, jobs):
    env = dict(os.environ, PYTHONPATH=f"{export}:{export / 'src'}")
    try:
        r = subprocess.run([PY, str(Path(__file__).resolve()), "--worker"], input=json.dumps(jobs),
                           capture_output=True, text=True, env=env, timeout=1800)
        return json.loads(r.stdout)
    except (OSError, ValueError, subprocess.SubprocessError) as e:
        return [dict(id=j["id"], ok=False, error=f"worker failed: {type(e).__name__}") for j in jobs]


def worker_main():
    """Runs inside an export's environment. stdin: jobs; stdout: results. Offline, no model."""
    jobs = json.loads(sys.stdin.read())
    out = []
    try:
        from pydantic import TypeAdapter

        from node_dag.factory import NodeConfig, build
        from node_dag.types import Dna

        adapter = TypeAdapter(NodeConfig)
    except Exception as e:  # noqa: BLE001
        print(json.dumps([dict(id=j["id"], ok=False, error=f"import failed: {e!r}"[:200]) for j in jobs]))
        return
    for j in jobs:
        try:
            if j["type"] == "gen":
                cfg = {k: v for k, v in j["config"].items() if k != "config_hash"}
                node = build(adapter.validate_python(cfg))
                pairs = []
                for i, seq in enumerate(j["inputs"]):
                    for o in node.run([Dna(sequence=seq)]):
                        pairs.append([i, o.sequence])
                out.append(dict(id=j["id"], ok=True, pairs=pairs))
            elif j["type"] == "gate":
                from node_dag.benchmark import Instance, gate, passes

                inst = Instance(key=j["gene"], parent=Dna(sequence=j["parent"]))
                rows = []
                for s in j["candidates"]:
                    gt = gate(inst, s)
                    rows.append(dict(gates=gt, passes=bool(passes(gt))))
                out.append(dict(id=j["id"], ok=True, rows=rows))
            else:
                out.append(dict(id=j["id"], ok=False, error="unknown job"))
        except Exception as e:  # noqa: BLE001
            out.append(dict(id=j["id"], ok=False, error=f"{type(e).__name__}: {e}"[:200]))
    print(json.dumps(out))

class Recoder:
    """Re-runs recorded generator steps over code exports; keeps the one that reproduces the bytes."""

    def __init__(self, shas, enabled=True):
        self.shas, self.enabled = list(shas), enabled
        self.jobs, self.result = {}, {}

    def add(self, jid, config, inputs, recorded):
        self.jobs[jid] = dict(type="gen", id=jid, config=config, inputs=inputs, recorded=recorded)

    def run(self):
        if not self.enabled:
            for jid in self.jobs:
                self.result[jid] = dict(ok=False, why="rerun disabled")
            return
        pending = dict(self.jobs)
        why = {}
        for sha in self.shas:
            if not pending:
                break
            ex = sha if isinstance(sha, Path) else ensure_export(sha)
            if ex is None:
                continue
            tag = ex.name
            res = run_worker(ex, [{k: v for k, v in j.items() if k != "recorded"} for j in pending.values()])
            for r in res:
                jid = r["id"]
                if not r.get("ok"):
                    why[jid] = f"{tag}: {r.get('error')}"
                    continue
                pool = {s for _, s in r["pairs"]}
                rec = set(pending[jid]["recorded"])
                if rec <= pool:
                    parents = {}
                    for i, s in r["pairs"]:
                        parents.setdefault(s, set()).add(pending[jid]["inputs"][i])
                    self.result[jid] = dict(ok=True, code=tag, parents=parents, exact=(rec == pool))
                    pending.pop(jid, None)
                else:
                    why[jid] = f"{tag}: re-run misses {len(rec - pool)} of {len(rec)} recorded outputs"
        for jid in pending:
            self.result[jid] = dict(ok=False, why=why.get(jid, "no code export available"))


def stop_changed(child: str, parent: str):
    """True/False when the parent ends in a stop codon, None when it has none to keep."""
    p = parent[-3:]
    if p not in STOPS:
        return None
    return child[-3:] != p


def gen_jobs(rid, dag, values):
    """Generator steps of this round whose input and output tables were both saved."""
    out = {}
    for k, st in steps_of(dag).items():
        cfg = g(st, "config", default={})
        if cfg.get("name") not in GEN:
            continue
        ins = g(st, "inputs", default={})
        src = ins.get("sequence") if isinstance(ins, dict) else None
        out[k] = dict(
            jid=f"{rid}|{k}",
            config=cfg,
            src=src,
            inputs=[e["sequence"] for e in items_of(values, src)] if src else [],
            recorded=[e["sequence"] for e in items_of(values, k)],
        )
    return out


def stop_check(kept, dag, values, gens, recoder):
    """-> dict(status, n, diff, ambiguous, same, nostop, unmeasurable, root_diff, detail)."""
    if not kept:
        return dict(status="n/a", detail="nothing kept")
    if any(e.get("kind", "dna") != "dna" for e in kept):
        return dict(status="n/a", detail="not DNA")
    inputs_keys = list(g(dag, "inputs", default={}) or {})
    input_ids = {e.get("id") for k in inputs_keys for e in items_of(values, k)}
    table_ids = {k: {e.get("id") for e in items_of(values, k)} for k in gens}
    seq_of_id = {}
    for k in list(gens) + inputs_keys:
        for e in items_of(values, k):
            seq_of_id.setdefault(e.get("id"), e["sequence"])

    def parents_of(seq, eid, depth=0):
        """(set of immediate parent seqs | None, codes used)."""
        if eid in input_ids:
            return {seq}, set()
        par, codes, bad = set(), set(), False
        cands = [k for k in gens if eid in table_ids[k]]
        if not cands:
            return None, set()
        for k in cands:
            r = recoder.result.get(gens[k]["jid"], {})
            if not r.get("ok"):
                bad = True
                continue
            pp = r["parents"].get(seq)
            if pp is None:
                bad = True
                continue
            par |= pp
            codes.add(r["code"])
        return (par if par else None), codes

    def root_of(seq, eid, depth=0):
        if eid in input_ids or depth > 6:
            return {seq}
        pp, _ = parents_of(seq, eid)
        if not pp:
            return None
        roots = set()
        for p in pp:
            pid = hashlib.sha256(f"dna:{p}".encode()).hexdigest()[:12]
            r = {p} if pid in input_ids else root_of(p, pid, depth + 1)
            if r is None:
                return None
            roots |= r
        return roots

    c = dict(diff=0, ambiguous=0, same=0, nostop=0, unmeasurable=0, root_diff=0)
    codes_all, chain = set(), False
    for e in kept:
        eid, seq = e.get("id"), e["sequence"]
        pp, codes = parents_of(seq, eid)
        if not pp:
            c["unmeasurable"] += 1
            continue
        codes_all |= codes
        flags = [stop_changed(seq, p) for p in pp]
        known = [f for f in flags if f is not None]
        if not known:
            c["nostop"] += 1
        elif all(known):
            c["diff"] += 1
        elif any(known):
            c["ambiguous"] += 1
        else:
            c["same"] += 1
        rt = root_of(seq, eid)
        if rt is not None and rt != pp:
            chain = True
            rflags = [stop_changed(seq, p) for p in rt]
            rk = [f for f in rflags if f is not None]
            if rk and all(rk):
                c["root_diff"] += 1
    n = len(kept)
    if c["unmeasurable"] == n:
        status = "unmeasurable"
    elif c["diff"]:
        status = "FAIL"
    elif c["ambiguous"]:
        status = "ambiguous"
    elif c["unmeasurable"]:
        status = "partial"
    else:
        status = "ok"
    c.update(status=status, n=n, chain=chain, code=",".join(sorted(codes_all)))
    return c


# ------------------------------------------------------------------------------ gate


def gene_of(h, goal):
    """Dev gene name when this is the benchmark CAI goal over a dev gene, else None / reason."""
    if not isinstance(goal, str) or not CAI_GOAL.search(goal):
        return None, None
    inp = h.get("inputs")
    seqs = []
    if isinstance(inp, dict):
        for v in inp.values():
            seqs += v if isinstance(v, list) else [v]
    if len(seqs) != 1 or not isinstance(seqs[0], dict):
        return None, "CAI goal but not a single input"
    seq = seqs[0].get("sequence")
    if not isinstance(seq, str):
        return None, "CAI goal but the input has no sequence"
    sha = hashlib.sha256(seq.encode()).hexdigest()[:12]
    gene = DEV_GENES.get(sha)
    return (gene, seq) if gene else (None, "CAI goal over a gene not in the dev table; gate not run")

def build_row(d, h, hid, a, v, acc, gene, gparent, recoder):
    rid = f"{hid}|r{a.get('round')}"
    dag, values, note = attempt_outcome(d, a, hid)
    kept, kkeys, how = kept_entities(dag, values, a.get("plan"))
    gens = gen_jobs(rid, dag, values)
    for gj in gens.values():
        if gj["inputs"] and gj["recorded"]:
            recoder.add(gj["jid"], gj["config"], gj["inputs"], gj["recorded"])
        else:
            recoder.result[gj["jid"]] = dict(ok=False, why="input or output table not saved")
    row = dict(round=a.get("round"), accepted=acc, verdict_ok=(v.get("agrees"), v.get("covers_goal")),
               tokens=round_tokens(d, hid, a.get("round")), n_kept=len(kept), kept_from=",".join(kkeys), how=how,
               have_outcome=bool(values), note=note, kept=kept, dag=dag, values=values, gens=gens,
               cols=[], any_first_claim=False)
    if values:
        def other_values():
            for o in h.get("attempts") or []:
                if isinstance(o, dict) and o is not a:
                    _, ov, _ = attempt_outcome(d, o, hid)
                    if ov:
                        yield ov

        row["cols"], row["any_first_claim"] = score_check(h, a.get("plan"), dag, values, kept, kkeys, other_values)
    row["score"], row["score_why"] = verdict_of_score(row["cols"], row["any_first_claim"], len(kept), bool(values), note)
    if gene and kept:
        row["gate_job"] = dict(type="gate", id=f"{rid}|gate", gene=gene, parent=gparent,
                               candidates=[e["sequence"] for e in kept])
    return row


def short_id(hid):
    m = re.match(r"hypothesis-([0-9a-f]{8})-", hid or "")
    return m.group(1) if m else re.sub(r"^hypothesis-", "", hid or "")[:24]


def collect(root, rounds, recoder):
    """Phase 1: read everything, queue re-run and gate jobs. Returns run dicts (rounds partial)."""
    index = load_index(root)
    runs = []
    for d in find_result_dirs(root):
        ledger = read_ledger(d)
        group = str(d.relative_to(Path(root))) if d != Path(root) else "."
        files = sorted(glob.glob(str(d / "hypotheses" / "*.json")))
        for f in files:
            h, why = jload(f)
            hid_guess = Path(f).stem
            if not isinstance(h, dict):
                runs.append(dict(group=group, id=hid_guess, short=short_id(hid_guess), state="unreadable", why=why,
                                 rounds=0, tokens=None, accepted=False, rows=[], kind="?"))
                continue
            hid = h.get("id") or hid_guess
            led = ledger.get(hid, {})
            ix = index.get(hid, {})
            atts = [a for a in (h.get("attempts") or []) if isinstance(a, dict)]
            usage = h.get("usage") if isinstance(h.get("usage"), dict) else {}
            goal = h.get("goal")
            gene, gparent = gene_of(h, goal)
            run = dict(
                group=group, id=hid, short=short_id(hid),
                stop_claimed=any(isinstance(c, dict) and "stop codon" in str(c.get("claim", "")).lower() for c in (h.get("criteria") or [])),
                state=h.get("state"), rounds=len(atts) or h.get("round") or 0,
                tokens=usage.get("total") or led.get("tokens"),
                usage={k: usage.get(k) for k in ("plan", "verify", "critique") if usage.get(k) is not None},
                why=h.get("stopped_because"), in_ledger=bool(led), kind=ix.get("kind") or ("?" if not index else "unlisted"),
                goal_key=ix.get("goal_key"), models=led.get("models"), commit=ix.get("commit_short"),
                gene=gene if isinstance(gene, str) else None, gene_note=gparent if not gene else None,
                accepted=False, rows=[],
            )
            for a in atts:
                v = a.get("verdict") if isinstance(a.get("verdict"), dict) else {}
                acc = bool(v.get("achieved"))
                if acc:
                    run["accepted"] = True
                if not acc and rounds != "all":
                    continue
                try:
                    run["rows"].append(build_row(d, h, hid, a, v, acc, gene, gparent, recoder))
                except Exception as e:  # noqa: BLE001  a half-written or odd file must not stop the scoring
                    run["rows"].append(dict(round=a.get("round"), accepted=acc, tokens=None, n_kept=0, kept=[], dag={}, values={}, gens={},
                                            have_outcome=False, cols=[], any_first_claim=False, score="UNMEASURABLE",
                                            score_why=f"error reading the round: {type(e).__name__}: {e}"[:160], note="error"))
            runs.append(run)
    return runs


def finish(runs, recoder, gate_sha):
    gate_jobs = [r["gate_job"] for run in runs for r in run["rows"] if r.get("gate_job")]
    gate_res = {}
    if gate_jobs:
        ex = ensure_export(gate_sha) or ensure_export("822d2ea")
        if ex:
            for r in run_worker(ex, gate_jobs):
                gate_res[r["id"]] = r
    for run in runs:
        for r in run["rows"]:
            try:
                r["stop"] = stop_check(r["kept"], r["dag"], r["values"], r["gens"], recoder) if r["have_outcome"] else dict(status="unmeasurable", detail="no outcome saved")
            except Exception as e:  # noqa: BLE001
                r["stop"] = dict(status="unmeasurable", detail=f"error {type(e).__name__}")
            gj = r.get("gate_job")
            if gj:
                res = gate_res.get(gj["id"])
                if res and res.get("ok"):
                    rows = res["rows"]
                    fails = {}
                    for x in rows:
                        for k, v in x["gates"].items():
                            if (k in ("protein_unchanged", "length_unchanged", "immutable_unchanged") and v != 1.0) or (k == "targets_remaining" and v != 0.0):
                                fails[k] = fails.get(k, 0) + 1
                    r["gate"] = dict(status="PASS" if all(x["passes"] for x in rows) else "FAIL",
                                     n=len(rows), n_pass=sum(x["passes"] for x in rows), fails=fails)
                else:
                    r["gate"] = dict(status="unmeasurable", detail=(res or {}).get("error", "gate worker failed"))
            else:
                r["gate"] = None
            for k in ("kept", "dag", "values", "gens", "gate_job"):
                r.pop(k, None)
    return runs


def denominators(runs):
    def count(rs):
        acc = [(run, r) for run in rs for r in run["rows"] if r["accepted"]]
        d = dict(
            launched=len(rs),
            accepted_runs=sum(1 for run in rs if str(run.get("state")) == "achieved"),
            accepted_rounds=len(acc),
            score_pass=sum(1 for _, r in acc if r["score"] == "PASS"),
            score_fail=sum(1 for _, r in acc if r["score"] == "FAIL"),
            score_empty=sum(1 for _, r in acc if r["score"] == "EMPTY"),
            score_unmeasurable=sum(1 for _, r in acc if r["score"] == "UNMEASURABLE"),
            stop_applicable=sum(1 for _, r in acc if r["stop"]["status"] != "n/a"),
            stop_fail=sum(1 for _, r in acc if r["stop"]["status"] == "FAIL"),
            stop_fail_claimed=sum(1 for run, r in acc if r["stop"]["status"] == "FAIL" and run.get("stop_claimed")),
            stop_unmeasurable=sum(1 for _, r in acc if r["stop"]["status"] in ("unmeasurable", "partial", "ambiguous")),
            score_and_stop_pass=sum(1 for _, r in acc if r["score"] == "PASS" and r["stop"]["status"] == "ok"),
            gate_scored=sum(1 for _, r in acc if r.get("gate")),
            gate_pass=sum(1 for _, r in acc if r.get("gate") and r["gate"]["status"] == "PASS"),
            state_mismatch=sum(1 for run in rs if run["accepted"] != (str(run.get("state")) == "achieved")),
        )
        d["accepted_failing_a_code_check"] = sum(
            1 for _, r in acc if r["score"] in ("FAIL", "EMPTY") or r["stop"]["status"] == "FAIL" or (r.get("gate") and r["gate"]["status"] == "FAIL"))
        return d

    out = {"all": count(runs)}
    kinds = sorted({str(run.get("kind")) for run in runs})
    if len(kinds) > 1:
        for kind in kinds:
            out[f"kind={kind}"] = count([r for r in runs if str(r.get("kind")) == kind])
    return out


def fmt_tok(x):
    return f"{int(x):,}" if isinstance(x, (int, float)) else "-"


def stop_text(s):
    if s["status"] in ("FAIL", "ok", "partial", "ambiguous"):
        t = f"stop {s['status']} (differs {s['diff']} of {s['n']} kept"
        t += f", ambiguous {s['ambiguous']}" if s["ambiguous"] else ""
        t += f", unmeasurable {s['unmeasurable']}" if s["unmeasurable"] else ""
        t += f", vs chain root {s['root_diff']}" if s.get("chain") else ""
        return t + (f"; code {s['code']})" if s.get("code") else ")")
    return f"stop {s['status']}" + (f" ({s.get('detail')})" if s.get("detail") else "")


def col_text(c):
    base = "-" if c["baseline"] is None else format(c["baseline"], ".10g")
    t = f"{'*' if c['counted'] else ''}{c['column']} {'above' if c['up'] else 'below'} {base}: not-better {c['n_not_better']} of {c['n_kept']}"
    t += f", {'lowest' if c['up'] else 'highest'} kept {c['edge']:.10g}" if c["edge"] is not None else ""
    return t + (f" [baseline from {c['baseline_from']}]" if c.get("baseline_from") not in (None, "this round") else "")


def brief_line(run, r):
    s, gt = r["stop"], r.get("gate")
    cnt = [c for c in r["cols"] if c["counted"]]
    c = cnt[0] if cnt else None
    score = r["score"]
    if c is not None and c["baseline"] is not None:
        score += f" {c['n_not_better']}/{c['n_kept']} not {'above' if c['up'] else 'below'} {c['baseline']:.6g}" + (f" ({'lowest' if c['up'] else 'highest'} {c['edge']:.6g})" if c["edge"] is not None else "")
    elif r["score"] in ("UNMEASURABLE", "EMPTY"):
        score += f" [{r['score_why']}]"
    stop = s["status"] if s["status"] not in ("FAIL", "ok", "partial", "ambiguous") else f"{s['status']} {s['diff']}/{s['n']}" + (f"+{s['ambiguous']}amb" if s["ambiguous"] else "") + (f" {s['unmeasurable']}unm" if s["unmeasurable"] else "")
    gate = "-" if not gt else (f"{gt['status']} {gt['n_pass']}/{gt['n']}" if gt.get("n") else gt["status"])
    return (f"{run['short']:13} {(run.get('goal_key') or run.get('gene') or '')[:10]:10} {str(run['state'])[:12]:12} rd{run['rounds']} {fmt_tok(run['tokens']):>8} "
            f"r{r['round']} {'ACC' if r['accepted'] else 'veto'} kept {r['n_kept']:>3} | SCORE {score} | STOP {stop}{' (claimed)' if run.get('stop_claimed') else ''} | GATE {gate}")


def print_brief(runs, dens, rounds):
    for run in runs:
        rs = [r for r in run["rows"] if r["accepted"] or rounds == "all"]
        if not rs:
            print(f"{run['short']:13} {(run.get('goal_key') or run.get('gene') or '')[:10]:10} {str(run['state'])[:12]:12} rd{run['rounds']} {fmt_tok(run['tokens']):>8} no accepted round")
        for r in rs:
            print(brief_line(run, r))
    print()
    for name, d in dens.items():
        print(f"[{name}] launched {d['launched']} | accepted {d['accepted_rounds']} rounds/{d['accepted_runs']} runs | score PASS {d['score_pass']} FAIL {d['score_fail']} EMPTY {d['score_empty']} UNMEAS {d['score_unmeasurable']} | stop FAIL {d['stop_fail']} of {d['stop_applicable']} (unmeas {d['stop_unmeasurable']}) | PASS score+stop {d['score_and_stop_pass']} | gate {d['gate_pass']}/{d['gate_scored']} | failing any code check {d['accepted_failing_a_code_check']}")


def print_text(runs, dens, rounds):
    print("Columns marked * are counted toward the score check (a criterion compares with the first input); others are info.")
    for run in runs:
        gk = (run.get("goal_key") or run.get("gene") or "")[:10]
        flag = " !state/verdict differ" if run["accepted"] != (str(run.get("state")) == "achieved") else ""
        print(f"{run['short']:9} {run['group'][-30:]:30} {gk:10} state={str(run['state'])[:13]:13} rounds={run['rounds']} tokens={fmt_tok(run['tokens'])}"
              + (f" stop-claimed" if run.get("stop_claimed") else "") + flag)
        for r in [r for r in run["rows"] if r["accepted"] or rounds == "all"]:
            tag = "ACCEPT" if r["accepted"] else "veto"
            tok = f" round tokens {fmt_tok(r['tokens'])}" if r["tokens"] else ""
            print(f"    r{r['round']} {tag}{tok}; kept {r['n_kept']}")
            print(f"        score {r['score']}: {r['score_why']}" + (" [" + "; ".join(col_text(c) for c in r["cols"]) + "]" if r["cols"] else ""))
            print(f"        {stop_text(r['stop'])}")
            if r.get("gate"):
                gt = r["gate"]
                print(f"        gate {gt['status']}" + (f": {gt['n_pass']} of {gt['n']} kept pass; failing gates {gt['fails']}" if gt.get("n") else f" ({gt.get('detail')})"))
        if not [r for r in run["rows"] if r["accepted"]] and run.get("gene_note"):
            print(f"    ({run['gene_note']})")
    print()
    for name, d in dens.items():
        print(f"DENOMINATORS [{name}]")
        print(f"  launched {d['launched']}; accepted by verdict {d['accepted_rounds']} rounds ({d['accepted_runs']} runs with state achieved; {d['state_mismatch']} runs where state and verdict differ)")
        print(f"  of the {d['accepted_rounds']} accepted rounds, score check: PASS {d['score_pass']}, FAIL {d['score_fail']}, EMPTY {d['score_empty']}, UNMEASURABLE {d['score_unmeasurable']}")
        print(f"  stop identity over {d['stop_applicable']} accepted rounds with DNA kept: FAIL {d['stop_fail']} ({d['stop_fail_claimed']} of them with a criterion that names the stop codon), unmeasurable/partial/ambiguous {d['stop_unmeasurable']}")
        print(f"  pass the score check AND every kept stop equals its parent's: {d['score_and_stop_pass']}")
        print(f"  benchmark gate (PR #14, immutable honoured): scored {d['gate_scored']}, pass {d['gate_pass']}")
        print(f"  accepted rounds failing at least one code check (score FAIL or EMPTY, stop FAIL, gate FAIL): {d['accepted_failing_a_code_check']}")


def main(argv=None):
    if argv is None:
        argv = sys.argv[1:]
    if argv[:1] == ["--worker"]:
        return worker_main()
    ap = argparse.ArgumentParser(description="Code-only scorer for a loop results directory.")
    ap.add_argument("dir")
    ap.add_argument("--json", action="store_true", help="print the JSON record instead of the table")
    ap.add_argument("--brief", action="store_true", help="one line per accepted round, then one line per denominator set")
    ap.add_argument("--rounds", choices=("accepted", "all"), default="accepted")
    ap.add_argument("--no-rerun", action="store_true", help="skip re-running recoder configs (stop check unmeasurable)")
    ap.add_argument("--src", action="append", default=[], help="extra code export dir to try for re-runs (before the defaults)")
    a = ap.parse_args(argv)
    shas = [Path(s) for s in a.src] + list(DEFAULT_SHAS)
    recoder = Recoder(shas, enabled=not a.no_rerun)
    if not find_result_dirs(Path(a.dir)):
        print(f"score_run: no hypotheses/ directory under {a.dir} (not started yet?)", file=sys.stderr)
        return 2
    runs = collect(Path(a.dir), a.rounds, recoder)
    recoder.run()
    runs = finish(runs, recoder, GATE_SHA)
    dens = denominators(runs)
    if a.json:
        print(json.dumps(dict(dir=str(a.dir), denominators=dens, runs=runs), indent=1, default=str))
    elif a.brief:
        print_brief(runs, dens, a.rounds)
    else:
        print_text(runs, dens, a.rounds)
    return 0


if __name__ == "__main__":
    sys.exit(main())
